\chapter{Phân Loại Phương Tiện}
\label{ch:phanloai}

Chương này trình bày module phân loại phương tiện trong pipeline nhận diện. Module gồm ba bước nối tiếp:

\begin{enumerate}[leftmargin=*, label=(\arabic*)]
  \item \textbf{Loại xe} (ô tô, xe máy, xe tải): model tự huấn luyện, chạy trên cả khung hình, cho nhãn \texttt{car}/\texttt{motorbike}/\texttt{truck}.
  \item \textbf{Định vị xe}: model YOLO pretrained (COCO), không huấn luyện thêm, chỉ dùng để tìm và cắt vùng xe phục vụ bước~(3) và trả tọa độ khung xe; \textit{nhãn lớp COCO của bước này không tham gia quyết định loại xe}.
  \item \textbf{Kiểu dáng xe con} (chỉ chạy khi bước~(1) ra ``ô tô'' và bước~(2) cắt được vùng xe): model tự huấn luyện, ba lớp \texttt{Sedan}, \texttt{Gầm cao}, \texttt{Xe tải}.
\end{enumerate}

Bước~(1) là phần bổ sung sau khi tích hợp pipeline. Bản đầu tiên dùng thẳng nhãn lớp COCO của YOLO pretrained làm loại xe, gộp bước loại xe và bước định vị làm một. Sau khi phát hiện cách này sai nhiều trên ảnh cận cảnh kiểu camera cổng, đề tài huấn luyện riêng một model cho bước loại xe và tách nó khỏi bước định vị; lý do và bằng chứng đo được trình bày tại Phần~\ref{sec:phanloai-ketqua} và Phần~\ref{sec:phanloai-thaoluan}.

\textbf{Về phạm vi so với đề cương.} Đề cương quy định huấn luyện đủ bốn lớp (ô tô, xe máy, xe tải, xe buýt) và đánh giá sâu hai lớp chính. Sau khi làm lại dữ liệu, đề tài đã huấn luyện được model riêng có accuracy và F1 đo được cho ba trong bốn lớp: \texttt{car}, \texttt{motorbike}, \texttt{truck}. Lớp \texttt{bus} chưa có model riêng do dữ liệu hiện có không đủ ảnh xe khách cận cảnh đúng góc camera cổng; phương án mượn nhãn \texttt{bus} của YOLO pretrained đã được thử nhưng bỏ vì đo được nhãn đó sai (Phần~\ref{sec:phanloai-ketqua}). Khoảng cách với đề cương vì vậy thu hẹp từ bốn trên bốn lớp xuống còn một trên bốn lớp; mức độ hoàn thành lớp \texttt{bus} cần trao đổi thêm với giảng viên hướng dẫn.

\textbf{Lần huấn luyện đầu tiên mắc lỗi học tắt (shortcut learning).} Khi mỗi lớp chỉ lấy từ đúng một nguồn ảnh (\texttt{car} từ \texttt{carlong\_}, \texttt{motorbike} từ \texttt{greenpack\_}, \texttt{truck} từ \texttt{Dieu\_}/\texttt{Hung\_}), model đạt 100\% trên tập test riêng nhưng chỉ đúng 1 trên 13 khi kiểm tra trên ảnh phố thật ngoài tập train, gần như mọi ảnh đều bị đoán thành \texttt{truck}. Nguyên nhân là \texttt{truck} khi đó là lớp duy nhất có ảnh phong cách đường phố, nên model học đặc điểm nguồn ảnh và góc camera thay vì hình dáng xe. Đây là bài học phương pháp quan trọng: accuracy cao trên tập test không chứng minh được gì nếu tập test chia sẻ cùng thiên lệch nguồn với tập train.

\textbf{Cách sửa là trộn phong cách ảnh cho từng lớp}, buộc model học hình dáng xe thay vì học nguồn ảnh. \section{Kiến Trúc Mô Hình}
\label{sec:phanloai-kientruc}

\textbf{Bước loại xe.} So sánh hai kiến trúc fine-tune từ trọng số ImageNet, thay lớp cuối cho ba lớp \texttt{car}/\texttt{motorbike}/\texttt{truck}: ResNet18~\autocite{he2016resnet} với 11.178.051 tham số, và MobileNetV3-Small~\autocite{howard2019mobilenetv3} với 1.520.931 tham số. Đầu vào là \textbf{cả khung hình}, resize 224$\times$224, chuẩn hóa theo thống kê ImageNet. Việc dùng cả khung hình chứ không dùng vùng xe đã cắt là có chủ đích: dữ liệu huấn luyện của bước này là ảnh camera cổng nguyên khung (script chuẩn bị dữ liệu chép thẳng ảnh gốc, không cắt), nên đưa ảnh nguyên vào lúc suy luận mới đúng phân phối lúc train.

\textbf{Bước định vị xe.} Một model YOLO pretrained trên COCO~\autocite{jocher2023yolov8}, không huấn luyện thêm, dùng để tìm và cắt vùng xe cho bước kiểu dáng và trả tọa độ khung xe. Đề tài dùng cụ thể YOLOv8n, lấy bounding box xe có diện tích lớn nhất, cắt kèm biên 10\%. Nhãn lớp COCO của bước này (\texttt{car}/\texttt{motorcycle}/\texttt{bus}/\texttt{truck}) từng được dùng thẳng làm loại xe ở bản đầu tiên, nay không còn dùng làm nhãn nữa. Thiết kế cho phép thay phiên bản YOLO khác mà không đổi logic pipeline.

\textbf{Bước kiểu dáng.} So sánh hai kiến trúc như bước loại xe (ResNet18 và MobileNetV3-Small, nhẹ hơn khoảng 7,3 lần), fine-tune từ ImageNet, thay lớp cuối cho ba lớp. Đầu vào là \textbf{vùng xe đã cắt} 224$\times$224 (không phải cả khung hình), vì model này huấn luyện trên ảnh B5 cắt theo bounding box. Augmentation cho cả hai bước gồm random-resized-crop (scale từ 0,8 đến 1,0), lật ngang, xoay 10 độ và color jitter.

\subsection{Kiểm tra ngoài phân phối (out-of-distribution)}
\label{subsec:phanloai-ood}

Để xác nhận lỗi học tắt đã được sửa tận gốc chứ không chỉ che bằng cách thêm dữ liệu tương tự tập test, đề tài dựng một bộ kiểm tra cố định gồm 12 ảnh thật có nhãn đúng đã biết, không nằm trong bất kỳ tập train, valid hay test nào, giữ lại vĩnh viễn làm bài kiểm tra hồi quy (\texttt{src/ml/sanity\_check\_vehicle\_type\_ood.py}). Kết quả: ResNet18 đúng 12 trên 12 (100\%), MobileNetV3-Small đúng 11 trên 12 (92\%). So với 1 trên 13 của lần huấn luyện đầu tiên, đây là bằng chứng trực tiếp cho thấy việc trộn phong cách ảnh đã giải quyết đúng nguyên nhân gốc. Bài học phương pháp rút ra: khi nghi ngờ model học tắt, cách kiểm chứng đáng tin là một tập ngoài phân phối nhỏ nhưng chắc chắn tách biệt nguồn với tập train, không phải chia lại một tập test lớn hơn từ cùng nguồn dữ liệu.

\subsection{Đo lại ở mức pipeline}
\label{subsec:phanloai-pipeline}

Accuracy 98,66\% ở trên là của riêng model, đo bằng cách nạp thẳng ảnh vào model. Khi ghép vào pipeline thật, kết quả người dùng nhận được có thể khác hẳn. Bản ghép đầu tiên đặt bước định vị YOLO làm điều kiện chạy của model loại xe: nếu YOLO không tìm được xe thì bỏ qua, không phân loại. Chạy lại chính tập test loại xe qua pipeline đầy đủ cho thấy hậu quả: YOLOv8n pretrained COCO chỉ nhận ra xe trên 64 trên 149 ảnh, bỏ sót 85 ảnh (57\%) vì không quen ảnh cận cảnh kiểu camera cổng. Ví dụ cụ thể là một ảnh xe máy chụp từ phía sau bị YOLO đoán thành \texttt{person} (62,5\%) và \texttt{handbag} (18,4\%), không có lớp xe nào, trong khi model loại xe chạy thẳng trên đúng ảnh đó cho \texttt{motorbike} 99,9\%.

Cách sửa là bỏ hẳn sự phụ thuộc này: model loại xe chạy trên cả khung hình đúng như dữ liệu lúc train, YOLO chỉ còn phục vụ bước kiểu dáng. Điều kiện an toàn duy nhất còn giữ là phải có bằng chứng thật sự có xe trong khung hình (YOLO thấy xe \textit{hoặc} module biển số đọc được biển), tránh khung hình trống bị ép về một trong ba lớp. \section{Thảo Luận}
\label{sec:phanloai-thaoluan}

\textbf{Accuracy trên tập test nội bộ không đảm bảo hiệu năng thật.} Đây là lý do ban đầu đề tài chọn YOLO pretrained cho bước loại thô thay vì tự huấn luyện: lần huấn luyện đầu đạt accuracy rất cao trên tập test riêng nhưng chỉ đúng 1 trên 13 khi gặp ảnh ngoài phân phối, vì mỗi lớp chỉ lấy từ một nguồn ảnh. Bài học rút ra không phải ``model tự huấn luyện luôn kém hơn pretrained'', mà là độ đa dạng nguồn ảnh trong tập train quan trọng hơn số lượng ảnh hay accuracy đo được trên một tập test cùng nguồn. Sau khi trộn nguồn ảnh đúng cách, model tự huấn luyện đạt 100\% và 92\% trên bộ kiểm tra ngoài phân phối, tốt hơn hẳn phương án dùng thẳng nhãn COCO.

\textbf{Accuracy của model không phải accuracy người dùng nhận được.} Bài học ở một tầng khác (Phần~\ref{subsec:phanloai-pipeline}): một thành phần chưa được huấn luyện (bước định vị YOLO) từng đứng chặn đầu vào của model đã đánh giá, làm mất 57\% số ca dù model xử lý được. Nguyên tắc khi ghép nhiều model là không để một thành phần chưa đánh giá chặn đầu vào của thành phần đã có số liệu; nếu cần điều kiện, hãy chọn điều kiện dựa trên bằng chứng độc lập (ở đây là ``đọc được biển số'') thay vì dựa vào chính thành phần yếu.

\textbf{Về phạm vi so với đề cương.} Quyết định ban đầu dùng model pretrained cho cả bốn lớp có căn cứ thực nghiệm tại thời điểm đó nhưng khác chữ ``huấn luyện'' của đề cương. Sau khi sửa dữ liệu, đề tài đã có model riêng cho ba trên bốn lớp, thu hẹp đáng kể khoảng cách. Lớp \texttt{bus} chưa có model riêng vì thiếu ảnh xe khách cận cảnh đúng góc camera cổng; hướng khắc phục là thu thập thêm dữ liệu xe khách theo đúng quy trình đã dùng cho ba lớp kia.

\textbf{Hạn chế còn lại của bước định vị.} YOLO pretrained COCO bỏ sót xe trên 57\% ảnh camera cổng trong tập test vì chưa từng huấn luyện với góc chụp này. Sau khi tách bước loại xe ra khỏi bước định vị, hậu quả đã thu hẹp: loại xe và biển số vẫn cho kết quả đầy đủ, chỉ còn tọa độ khung xe (chỉ để hiển thị) và bước kiểu dáng (cần vùng crop) phụ thuộc bước định vị. Với 57\% ảnh nói trên, hệ thống vẫn ra loại xe và biển số đúng, chỉ thiếu thông tin kiểu dáng xe con.

\textbf{Hạn chế của bước kiểu dáng.} Model kiểu dáng vẫn huấn luyện hoàn toàn trên B5 (ảnh dealer và showroom). Con số 90\% trên tập test B5 phản ánh khả năng phân biệt kiểu dáng trong cùng phong cách ảnh với tập train, chưa chứng minh được trên ảnh camera giám sát Việt Nam thật; kết quả kiểm thử thực địa cho tín hiệu tích cực nhưng cỡ mẫu quá nhỏ để kết luận chắc.

\textbf{Chọn model triển khai.} Ở bước kiểu dáng, hai kiến trúc gần như ngang nhau về accuracy (0,8997 so với 0,9007) nhưng MobileNetV3-Small nhẹ hơn 7,3 lần và nhanh hơn gần hai lần trên CPU. Ở bước loại xe, ResNet18 chính xác hơn một chút (0,9866 so với 0,9799 trên tập test, và 100\% so với 92\% trên bộ ngoài phân phối), nên pipeline hiện dùng ResNet18 làm mặc định cho cả hai bước. Với mục tiêu chạy trên Raspberry Pi~5, MobileNetV3-Small vẫn là ứng viên đáng cân nhắc nếu độ trễ trên Pi là điểm nghẽn; quyết định cuối cùng chờ số đo thật trên Pi~5 (Chương~\ref{ch:trienkhai}).

\section{Kết Luận}
\label{sec:phanloai-ketluan}

Đề tài xây dựng thành công module phân loại phương tiện ba bước, đạt các kết quả sau:

\begin{itemize}[leftmargin=*]
\item Huấn luyện riêng model cho bước loại xe (\texttt{car}/\texttt{motorbike}/\texttt{truck}) thay cho việc dùng thẳng nhãn COCO của YOLO pretrained, đạt accuracy 0,9866 (ResNet18) và 0,9799 (MobileNetV3-Small), thu hẹp khoảng cách với đề cương từ bốn trên bốn xuống còn một trên bốn lớp.
\item Phát hiện và sửa một lỗi học tắt ở lần huấn luyện đầu (100\% test nhưng 1 trên 13 trên ảnh ngoài phân phối) bằng cách trộn nguồn ảnh cho mỗi lớp; dựng một bài kiểm tra ngoài phân phối cố định làm bài kiểm tra hồi quy lâu dài.
\item Đo lại chất lượng ở mức pipeline chứ không chỉ mức model, phát hiện và sửa lỗi bước định vị chặn 57\% đầu vào; sau khi sửa, tỉ lệ ảnh có kết quả tăng từ 43\% lên 100\% và độ chính xác end-to-end từ 41,6\% lên 98,7\%.
\item Gộp 12 kiểu dáng gốc của B5 thành ba nhóm thực dụng, huấn luyện và so sánh ResNet18 với MobileNetV3-Small cho bước kiểu dáng, đạt accuracy 0,8997 và 0,9007 kèm ma trận nhầm lẫn, đường hội tụ và số liệu tài nguyên đầy đủ.
\item Xuất cả hai model của cả hai bước sang ONNX và tích hợp vào pipeline dùng chung cho backend và edge worker (\texttt{src/ml/pipeline/onnx\_pipeline.py}), sẵn sàng cho bước triển khai thiết bị biên (Chương~\ref{ch:trienkhai}).
\end{itemize}

Ba điểm cần lưu ý khi đọc kết quả: lớp \texttt{bus} chưa có model riêng và không còn dùng nhãn COCO làm phương án tạm, nên xe khách bị xếp vào một trong ba lớp hiện có cho tới khi nhân viên sửa tay; model kiểu dáng huấn luyện hoàn toàn trên ảnh dealer nên con số 90\% chưa phản ánh hiệu năng trên ảnh camera giám sát thật; các số CPU inference đo trên máy huấn luyện, chưa phải Raspberry Pi~5. Cả ba điểm để lại cho các bước tiếp theo.
