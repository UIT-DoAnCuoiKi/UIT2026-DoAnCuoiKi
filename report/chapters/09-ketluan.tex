\chapter{Kết Luận và Hướng Phát Triển}
\label{ch:ketluan}

\section{Tóm Tắt Đóng Góp}
\wip{Tổng kết: hệ thống end-to-end khả chuyển, chạy biên; số liệu chính đạt được; bộ dữ liệu tự thu thập.}

\section{Sản Phẩm Bàn Giao}
\wip{Đối chiếu "kết quả mong đợi" (đề cương): phần mềm end-to-end + dashboard; mã nguồn;
bộ dữ liệu tự thu thập; video demo sản phẩm; báo cáo hoàn chỉnh.}

\section{Ứng Dụng Thực Tiễn}
\wip{Bãi giữ xe trường học/chung cư/thương mại; khả năng mở rộng.}

\section{Giới Hạn}
\wip{Nhắc lại giới hạn chính và mức độ ảnh hưởng.}

\section{Hướng Phát Triển}
\wip{Ngắn hạn (mở rộng dữ liệu, free-flow); trung/dài hạn (tích hợp thanh toán, đa cổng, đa camera).}
