\textbf{BiLSTM} & Bidirectional Long Short-Term Memory \\

\textbf{CLAHE} & Contrast Limited Adaptive Histogram Equalization \\
\textbf{CNN} & Convolutional Neural Network (mạng nơ ron tích chập) \\
\textbf{COCO} & Common Objects in Context (bộ dữ liệu) \\
\textbf{CPU} & Central Processing Unit \\
\textbf{CRNN} & Convolutional Recurrent Neural Network \\
\textbf{CSV} & Comma-Separated Values \\
\textbf{CTC} & Connectionist Temporal Classification \\
\textbf{FP32 / INT8} & Số thực 32 bit / số nguyên 8 bit (độ chính xác biểu diễn trọng số) \\
\textbf{GPIO} & General-Purpose Input/Output \\
\textbf{GPU} & Graphics Processing Unit \\
\textbf{HMAC} & Hash-based Message Authentication Code \\
\textbf{HSV} & Hue, Saturation, Value (không gian màu) \\
\textbf{IoU} & Intersection over Union \\
\textbf{JPEG} & Joint Photographic Experts Group (định dạng ảnh) \\
\textbf{JWT} & JSON Web Token \\
\textbf{mAP} & mean Average Precision \\
\textbf{OCR} & Optical Character Recognition (nhận dạng ký tự quang học) \\
\textbf{ONNX} & Open Neural Network Exchange \\
\textbf{PMIC} & Power Management Integrated Circuit \\
\textbf{RAM} & Random Access Memory \\
\textbf{RSS} & Resident Set Size (bộ nhớ thường trú của tiến trình) \\
\textbf{RTSP} & Real-Time Streaming Protocol \\
\textbf{SHA-256} & Secure Hash Algorithm 256 bit \\
\textbf{SoC} & System on a Chip \\
\textbf{YOLO} & You Only Look Once \\
