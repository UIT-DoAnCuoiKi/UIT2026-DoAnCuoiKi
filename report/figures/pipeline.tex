% Sơ đồ pipeline nhận diện phương tiện.
% Dựng theo src/ml/pipeline/onnx_pipeline.py phương thức run():
%   - nhánh biển (chạy trước): PlateDetector (YOLOv8n)
%                  -> process_plate (HSV + CLAHE + Otsu) -> màu nền
%                  -> read_plate (tách dòng) -> CRNN CTC (ONNX) -> chuỗi biển
%   - nhánh xe (chạy khi có bằng chứng xe: YOLO thấy xe HOẶC đã phát hiện biển):
%                  detect_vehicle_crop (YOLOv8n COCO) định vị + cắt vùng xe
%                  -> ResNet18 (ONNX) phân loại LOẠI xe (chạy trên cả khung hình)
%                  -> nếu loại xe == ô tô: MobileNetV3-Small (ONNX) phân loại KIỂU DÁNG
% Biên dịch riêng: xelatex pipeline.tex  (xuất pipeline.pdf, rồi sips ra .png)
\documentclass[border=6pt]{standalone}
\usepackage{fontspec}
\setmainfont{Times New Roman}
\usepackage{tikz}
\usetikzlibrary{positioning,arrows.meta,shapes.geometric,calc}

\begin{document}
\begin{tikzpicture}[
  font=\small,
  >={Stealth[round]},
  box/.style={rectangle, rounded corners=2pt, draw=black!70, fill=blue!5,
              align=center, minimum height=11mm, inner sep=4pt, text width=32mm},
  dec/.style={diamond, aspect=2.2, draw=black!70, fill=orange!12,
              align=center, inner sep=0pt, text width=18mm},
  outbox/.style={rectangle, rounded corners=2pt, draw=black!70, fill=green!8,
              align=center, minimum height=14mm, inner sep=4pt, text width=32mm},
  arr/.style={->, thick, draw=black!75},
  node distance=9mm and 12mm,
]

\node[box, fill=black!4] (in) {Khung ảnh BGR};

% nhánh biển (trên) — chạy trước
\node[box, above right=7mm and 14mm of in] (pdet) {Phát hiện biển số\\{\footnotesize YOLOv8n}};
\node[box, right=of pdet] (proc) {Nhận màu nền\\{\footnotesize HSV + CLAHE + Otsu}\\$\rightarrow$ màu nền};
\node[box, right=of proc] (ocr) {Đọc biển\\{\footnotesize tách dòng, CRNN CTC}\\$\rightarrow$ chuỗi biển};

% nhánh xe (dưới) — chạy khi có bằng chứng xe
\node[box, below right=7mm and 14mm of in] (vdet) {Định vị + cắt vùng xe\\{\footnotesize YOLOv8n COCO}};
\node[box, right=of vdet] (vtype) {Loại xe\\{\footnotesize ResNet18, ONNX}};
\node[dec, right=of vtype] (iscar) {loại xe\\= ô tô?};
\node[box, right=of iscar] (style) {Kiểu dáng\\{\footnotesize MobileNetV3-Small, ONNX}};

% tổng hợp kết quả, đặt bên phải, giữa hai nhánh
\node[outbox] (res) at ($(ocr.east)!0.5!(style.east) + (30mm,0)$)
  {Kết quả mỗi xe:\\chuỗi biển, màu nền,\\loại xe, kiểu dáng};

\draw[arr] (in.north) |- (pdet.west);
\draw[arr] (in.south) |- (vdet.west);
\draw[arr] (pdet) -- (proc);
\draw[arr] (proc) -- (ocr);
\draw[arr] (vdet) -- (vtype);
\draw[arr] (vtype) -- (iscar);
\draw[arr] (iscar) -- node[above, font=\footnotesize] {có} (style);

\draw[arr] (ocr.east) -| (res.north);
\draw[arr] (style.east) -| (res.south);
\draw[arr] (iscar.south) to[out=-90,in=180] node[pos=0.42, below, font=\footnotesize] {không} (res.west);

\end{tikzpicture}
\end{document}
